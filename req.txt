import serial
import time
import sys
import glob

def find_serial_port():
    ports = glob.glob('/dev/cu.usbmodem*')
    return ports[0] if ports else None

def test_esp32():
    port = find_serial_port()
    if not port:
        print("❌ No ESP32 (usbmodem) found on Mac.")
        return

    print(f"🔌 Connecting to ESP32 on {port}...")
    try:
        ser = serial.Serial(port, 115200, timeout=1)
        time.sleep(2) # Wait for reset
        
        # Flush buffers
        ser.reset_input_buffer()
        
        print("💡 Testing LED ON (l 1)...")
        ser.write(b'l 1\n')
        line = ser.readline().decode().strip()
        print(f"📥 Response: {line}")
        time.sleep(1)

        print("💡 Testing LED OFF (l 0)...")
        ser.write(b'l 0\n')
        line = ser.readline().decode().strip()
        print(f"📥 Response: {line}")
        time.sleep(1)

        print("🏁 Testing Motor Command pulse (m 0 0)...")
        ser.write(b'm 0.0 0.0\n')
        
        # Check for telemetry
        print("📡 Monitoring Telemetry (3s)...")
        start = time.time()
        while time.time() - start < 3:
            if ser.in_waiting:
                print(f"📥 {ser.readline().decode().strip()}")
        
        ser.close()
        print("✅ Test Complete.")

    except Exception as e:
        print(f"❌ Error: {e}")

if __name__ == "__main__":
    test_esp32()
